           Summary of the layered approach
           - Domain layer (core game rules, entities)
             - Snake, Food, Position, Direction, Board, GameState
             - All business logic (move, grow, collisions, game over, scoring)
             - Independent of rendering or input methods
           - Application layer (use cases / services)
             - GameService coordinates domain objects, handles game rules, and exposes a simple API (start, update, change direction)
             - DTOs or simple data structures for communication
           - Infrastructure layer (adapters)
             - Rendering adapter (e.g., Pygame) that translates domain state into visuals
             - Input adapter to map user input into domain actions
             - Persistence (e.g., high scores) if you want to store data
           - Presentation layer (UI/bootstrap)
             - Wires the app together: creates the service, renderer, and input adapter; runs the game loop

           from enum import Enum
           from dataclasses import dataclass
           from typing import List
           import random
           
           class Direction(Enum):
               UP = (0, -1)
               DOWN = (0, 1)
               LEFT = (-1, 0)
               RIGHT = (1, 0)
           
               def delta(self):
                   return self.value
           
           @dataclass(frozen=True)
           class Position:
               x: int
               y: int
           
           @dataclass
           class Snake:
               body: List[Position]
               direction: Direction
           
               def head(self) -> Position:
                   return self.body[0]
           
               def move(self, grow: bool = False) -> None:
                   dx, dy = self.direction.delta()
                   new_head = Position(self.head().x + dx, self.head().y + dy)
                   self.body.insert(0, new_head)
                   if not grow:
                       self.body.pop()
           
               def set_direction(self, new_dir: Direction) -> None:
                   opposite = {
                       Direction.UP: Direction.DOWN,
                       Direction.DOWN: Direction.UP,
                       Direction.LEFT: Direction.RIGHT,
                       Direction.RIGHT: Direction.LEFT,
                   }
                   if new_dir != opposite[self.direction]:
                       self.direction = new_dir
           
               def collides_with_self(self) -> bool:
                   return self.head() in self.body[1:]
           
           @dataclass
           class Food:
               position: Position
           
           @dataclass
           class GameBoard:
               width: int
               height: int
           
               def inside(self, pos: Position) -> bool:
                   return 0 <= pos.x < self.width and 0 <= pos.y < self.height
           
           from enum import Enum
           class GameState(Enum):
               READY = 0
               RUNNING = 1
               GAME_OVER = 2

           3) application/game_service.py
           - The application service orchestrates domain entities and exposes a clean API.
           
           import random
           from typing import Optional
           from domain.models import Position, Direction, Snake, Food, GameBoard, GameState
           
           class GameService:
               def __init__(self, width: int, height: int):
                   self.board = GameBoard(width, height)
                   mid = Position(width // 2, height // 2)
                   init_length = 3
                   init_body = [Position(mid.x - i, mid.y) for i in range(init_length)]
                   self.snake = Snake(body=init_body, direction=Direction.RIGHT)
                   self.food = self._place_food()
                   self.state = GameState.READY
                   self.score = 0
           
               def _place_food(self) -> Food:
                   occupied = set(self.snake.body)
                   while True:
                       pos = Position(random.randrange(self.board.width),
                                      random.randrange(self.board.height))
                       if pos not in occupied:
                           return Food(pos)
           
               def start_game(self) -> None:
                   self.state = GameState.RUNNING
           
               def update(self) -> None:
                   if self.state != GameState.RUNNING:
                       return
           
                   head = self.snake.head()
                   dx, dy = self.snake.direction.delta()
                   next_head = Position(head.x + dx, head.y + dy)
           
                   # Wall collision
                   if not self.board.inside(next_head):
                       self.state = GameState.GAME_OVER
                       return
           
                   # Self collision
                   if next_head in self.snake.body:
                       self.state = GameState.GAME_OVER
                       return
           
                   # Food check
                   grow = next_head == self.food.position
                   self.snake.move(grow=grow)
           
                   if grow:
                       self.score += 1
                       self.food = self._place_food()
           
               def change_direction(self, new_dir: Direction) -> None:
                   self.snake.set_direction(new_dir)
           
               # Simple getters for the UI / rendering layer
               def get_state(self):
                   return {
                       "state": self.state,
                       "score": self.score,
                       "snake": self.snake,
                       "food": self.food,
                   }
           
           4) infrastructure/rendering/pygame_renderer.py
           - Adapter to render domain state using Pygame.
           
           import pygame
           from domain.models import Snake, Food, Position, GameState
           
           class PygameRenderer:
               def __init__(self, cell_size: int = 20, margin: int = 2):
                   self.cell_size = cell_size
                   self.margin = margin
                   self.screen = None
           
               def init_display(self, width: int, height: int):
                   pygame.init()
                   self.screen = pygame.display.set_mode((width, height))
                   pygame.display.set_caption("Snake (Layered Architecture)")
           
               def render(self, snake: Snake, food: Food, state: GameState, width: int, height: int):
                   if self.screen is None:
                       self.init_display(width * self.cell_size, height * self.cell_size)
           
                   self.screen.fill((0, 0, 0))  # black background
           
                   # Draw snake
                   for idx, part in enumerate(snake.body):
                       color = (0, 255, 0) if idx == 0 else (0, 180, 0)
                       self._draw_cell(part.x, part.y, color)
           
                   # Draw food
                   self._draw_cell(food.position.x, food.position.y, (255, 0, 0))
           
                   # Optionally draw state text
                   if state == GameState.GAME_OVER:
                       self._draw_text("Game Over", (255, 255, 255), (10, 10))
           
                   pygame.display.flip()
           
               def _draw_cell(self, x: int, y: int, color: tuple):
                   rect = pygame.Rect(
                       x * self.cell_size,
                       y * self.cell_size,
                       self.cell_size - self.margin,
                       self.cell_size - self.margin,
                   )
                   pygame.draw.rect(self.screen, color, rect)
           
               def _draw_text(self, text: str, color: tuple, position: tuple):
                   font = pygame.font.SysFont(None, 24)
                   img = font.render(text, True, color)
                   self.screen.blit(img, position)
           
           5) infrastructure/input/pygame_input.py
           - Simple input adapter translating key presses to domain actions.
           
           import pygame
           from domain.models import Direction
           from application.game_service import GameService
           
           class PygameInput:
               def __init__(self, service: GameService):
                   self.service = service
           
               def handle_events(self):
                   for event in pygame.event.get():
                       if event.type == pygame.QUIT:
                           return False  # signal to quit
                       if event.type == pygame.KEYDOWN:
                           if event.key == pygame.K_UP:
                               self.service.change_direction(Direction.UP)
                           elif event.key == pygame.K_DOWN:
                               self.service.change_direction(Direction.DOWN)
                           elif event.key == pygame.K_LEFT:
                               self.service.change_direction(Direction.LEFT)
                           elif event.key == pygame.K_RIGHT:
                               self.service.change_direction(Direction.RIGHT)
                   return True
           
           6) presentation/main.py
           - Bootstrap and game loop (mixes layers in a simple runner).
           
           import pygame
           from application.game_service import GameService
           from infrastructure.rendering.pygame_renderer import PygameRenderer
           from infrastructure.input.pygame_input import PygameInput
           from domain.models import GameState
           
           def main():
               width, height = 30, 20  # grid size
               service = GameService(width, height)
               renderer = PygameRenderer(cell_size=20)
               input_adapter = PygameInput(service)
           
               # Start the game
               service.start_game()
           
               running = True
               clock = pygame.time.Clock()
               while running:
                   # Handle input
                   running = input_adapter.handle_events()
           
                   # Update game logic
                   service.update()
           
                   # Render
                   state = service.get_state()
                   snake = state["snake"]
                   food = state["food"]
                   renderer.render(snake, food, state["state"], width, height)
           
                   clock.tick(10)  # FPS can be tweaked
           
           if __name__ == "__main__":
               main()
           
           Notes and design considerations
           - Why a layered approach helps:
             - Domain layer has the game rules and data structures; you can unit-test snakes, collisions, and scoring without any UI.
             - Application layer coordinates actions and enforces use-case transitions (READY -> RUNNING -> GAME_OVER).
             - Infrastructure adapters decouple rendering and input from the domain, making it easier to swap in another UI (e.g., Tkinter, a web UI) later.
           - Rendering choice:
             - The above uses Pygame, which is common for 2D games in Python. If you prefer a terminal-based version, you could implement a small ASCII renderer in the infrastructure/rendering layer and still keep domain/app layers intact.
           - Persistence:
             - If you want high scores, add a simple repository in infrastructure/persistence (e.g., highscore_repo.py) and expose save/load methods. The domain or application layer should depend on an abstract interface (e.g., HighScoreRepository) rather than a concrete implementation to stay testable.
           - Testing tips:
             - Write unit tests for domain: ensure that moving the snake, growing on food, self-collision, and wall-collision behave as expected.
             - Mock the rendering layer when testing GameService to verify state changes without a real UI.

           import json
           from pathlib import Path
           
           class HighScoreRepository:
               def __init__(self, path: str = 'highscores.json'):
                   self.path = Path(path)
                   if not self.path.exists():
                       self.path.write_text('[]')
           
               def save(self, score: int):
                   scores = self.load_all()
                   scores.append(score)
                   self.path.write_text(json.dumps(scores))
           
               def load_all(self):
                   try:
                       data = json.loads(self.path.read_text())
                       if isinstance(data, list):
                           return data
                       return []
                   except Exception:
                       return []

           import random
           from typing import Optional
           from snake_project.domain.models import Position, Direction, Snake, Food, GameBoard, GameState
           
           class GameService:
               def __init__(self, width: int, height: int):
                   self.board = GameBoard(width, height)
                   mid = Position(width // 2, height // 2)
                   init_length = 3
                   init_body = [Position(mid.x - i, mid.y) for i in range(init_length)]
                   self.snake = Snake(body=init_body, direction=Direction.RIGHT)
                   self.food = self._place_food()
                   self.state = GameState.READY
                   self.score = 0
           
               def _place_food(self) -> Food:
                   occupied = set(self.snake.body)
                   while True:
                       pos = Position(random.randrange(self.board.width),
                                      random.randrange(self.board.height))
                       if pos not in occupied:
                           return Food(pos)
           
               def start_game(self) -> None:
                   self.state = GameState.RUNNING
           
               def update(self) -> None:
                   if self.state != GameState.RUNNING:
                       return
           
                   head = self.snake.head()
                   dx, dy = self.snake.direction.delta()
                   next_head = Position(head.x + dx, head.y + dy)
           
                   # Wall collision
                   if not self.board.inside(next_head):
                       self.state = GameState.GAME_OVER
                       return
           
                   # Self collision
                   if next_head in self.snake.body:
                       self.state = GameState.GAME_OVER
                       return
           
                   # Food check
                   grow = next_head == self.food.position
                   self.snake.move(grow=grow)
           
                   if grow:
                       self.score += 1
                       self.food = self._place_food()
           
               def change_direction(self, new_dir: Direction) -> None:
                   self.snake.set_direction(new_dir)
           
               # Simple getters for the UI / rendering layer
               def get_state(self):
                   return {
                       "state": self.state,
                       "score": self.score,
                       "snake": self.snake,
                       "food": self.food,
                   }

           import pygame
           from snake_project.domain.models import Snake, Food, Position, GameState
           
           class PygameRenderer:
               def __init__(self, cell_size: int = 20, margin: int = 2):
                   self.cell_size = cell_size
                   self.margin = margin
                   self.screen = None
           
               def init_display(self, width: int, height: int):
                   pygame.init()
                   self.screen = pygame.display.set_mode((width, height))
                   pygame.display.set_caption("Snake (Layered Architecture)")
           
               def render(self, snake: Snake, food: Food, state: GameState, width: int, height: int):
                   if self.screen is None:
                       self.init_display(width * self.cell_size, height * self.cell_size)
           
                   self.screen.fill((0, 0, 0))  # black background
           
                   # Draw snake
                   for idx, part in enumerate(snake.body):
                       color = (0, 255, 0) if idx == 0 else (0, 180, 0)
                       self._draw_cell(part.x, part.y, color)
           
                   # Draw food
                   self._draw_cell(food.position.x, food.position.y, (255, 0, 0))
           
                   # Optionally draw state text
                   if state == GameState.GAME_OVER:
                       self._draw_text("Game Over", (255, 255, 255), (10, 10))
           
                   pygame.display.flip()
           
               def _draw_cell(self, x: int, y: int, color: tuple):
                   rect = pygame.Rect(
                       x * self.cell_size,
                       y * self.cell_size,
                       self.cell_size - self.margin,
                       self.cell_size - self.margin,
                   )
                   pygame.draw.rect(self.screen, color, rect)
           
               def _draw_text(self, text: str, color: tuple, position: tuple):
                   font = pygame.font.SysFont(None, 24)
                   img = font.render(text, True, color)
                   self.screen.blit(img, position)

           import pygame
           from snake_project.domain.models import Direction
           
           class PygameInput:
               def __init__(self, service):
                   self.service = service
           
               def handle_events(self):
                   for event in pygame.event.get():
                       if event.type == pygame.QUIT:
                           return False  # signal to quit
                       if event.type == pygame.KEYDOWN:
                           if event.key == pygame.K_UP:
                               self.service.change_direction(Direction.UP)
                           elif event.key == pygame.K_DOWN:
                               self.service.change_direction(Direction.DOWN)
                           elif event.key == pygame.K_LEFT:
                               self.service.change_direction(Direction.LEFT)
                           elif event.key == pygame.K_RIGHT:
                               self.service.change_direction(Direction.RIGHT)
                   return True
