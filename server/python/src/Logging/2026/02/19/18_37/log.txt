           """Piece model for chess MVP.
           Single responsibility: represent a chess piece with color and kind.
           """
           
           from enum import Enum
           
           class Color(Enum):
               WHITE = 'w'
               BLACK = 'b'
           
           class Piece:
               def __init__(self, color: Color, kind: str):
                   self.color = color
                   self.kind = kind.upper()  # one of 'K','Q','R','B','N','P'
               def __repr__(self):
                   return f"{self.color.value}{self.kind}"
               def __str__(self):
                   return self.__repr__()
           
           __all__ = ['Color', 'Piece']

           """
           Board representation for chess MVP.
           Responsibilities:
           - Store an 8x8 grid with Piece instances or None.
           - Initialize the standard starting position.
           - Small helpers to get/set pieces by (row, col).
           """
           
           from typing import Optional, List
           from .pieces import Piece, Color
           
           class Board:
               SIZE = 8
           
               def __init__(self):
                   self.grid: List[List[Optional[Piece]]] = [[None for _ in range(self.SIZE)] for _ in range(self.SIZE)]
           
               def setup_starting_position(self):
                   # Black major pieces on row 0, pawns on row 1
                   back_rank = [Piece(Color.BLACK, 'R'), Piece(Color.BLACK, 'N'), Piece(Color.BLACK, 'B'),
                                Piece(Color.BLACK, 'Q'), Piece(Color.BLACK, 'K'), Piece(Color.BLACK, 'B'),
                                Piece(Color.BLACK, 'N'), Piece(Color.BLACK, 'R')]
                   self.grid[0] = back_rank
                   self.grid[1] = [Piece(Color.BLACK, 'P') for _ in range(self.SIZE)]
                   # White pawns on row 6, back rank on row 7
                   self.grid[6] = [Piece(Color.WHITE, 'P') for _ in range(self.SIZE)]
                   white_back = [Piece(Color.WHITE, 'R'), Piece(Color.WHITE, 'N'), Piece(Color.WHITE, 'B'),
                                 Piece(Color.WHITE, 'Q'), Piece(Color.WHITE, 'K'), Piece(Color.WHITE, 'B'),
                                 Piece(Color.WHITE, 'N'), Piece(Color.WHITE, 'R')]
                   self.grid[7] = white_back
           
               def get_piece(self, row: int, col: int) -> Optional[Piece):
                   return self.grid[row][col]
           
               def set_piece(self, row: int, col: int, piece: Optional[Piece]):
                   self.grid[row][col] = piece
           
               def clone(self) -> 'Board':
                   new_board = Board()
                   for r in range(self.SIZE):
                       new_board.grid[r] = [self.grid[r][c] if self.grid[r][c] is None else Piece(self.grid[r][c].color, self.grid[r][c].kind) for c in range(self.SIZE)]
                   return new_board
           
               def __repr__(self):
                   lines = []
                   for r in range(self.SIZE):
                       line = []
                       for c in range(self.SIZE):
                           p = self.grid[r][c]
                           line.append(str(p) if p else '.')
                       lines.append(' '.join(line))
                   return '\n'.join(lines)

           """
           Console-based UI for Chess MVP.
           - Renders the board using ASCII characters.
           - Parses user input like 'e2e4' into a Move.
           """
           
           from .board import Board
           from .move import Move
           from .utils import algebraic_to_coords, coords_to_algebraic
           from .pieces import Color, Piece
           
           
           def render_board(board: Board):
               # Print a simple ASCII board with coordinates
               print("  a b c d e f g h")
               for row in range(8):
                   line = []
                   for col in range(8):
                       piece = board.get_piece(row, col)
                       line.append(str(piece) if piece else '.')
                   print(f"{8 - row} " + ' '.join(line) + f" {8 - row}")
               print("  a b c d e f g h")
           
           
           def parse_move_input(input_str: str) -> Move:
               # Expecting something like 'e2e4' or 'e7e8q' for promotion
               s = input_str.strip()
               if len(s) < 4:
                   raise ValueError("Invalid move format")
               from_file = s[0]
               from_rank = s[1]
               to_file = s[2]
               to_rank = s[3]
               from_col = ord(from_file) - ord('a')
               from_row = 8 - int(from_rank)
               to_col = ord(to_file) - ord('a')
               to_row = 8 - int(to_rank)
               promo = s[4].upper() if len(s) >= 5 else None
               return Move(from_row, from_col, to_row, to_col, promotion=promo)
           
           
           def prompt_move() -> Move:
               user_input = input("Enter your move (e.g., e2e4): ")
               return parse_move_input(user_input)

           """
           Utility helpers for coordinate conversions and simple formatting.
           """
           
           
           def algebraic_to_coords(square: str):
               file = square[0].lower()
               rank = square[1]
               col = ord(file) - ord('a')
               row = 8 - int(rank)
               return row, col
           
           
           def coords_to_algebraic(row: int, col: int) -> str:
               file = chr(ord('a') + col)
               rank = str(8 - row)
               return f"{file}{rank}"
           
           __all__ = ['algebraic_to_coords', 'coords_to_algebraic']

           """Piece model for chess MVP.
           Single responsibility: represent a chess piece with color and kind.
           This is a simple data container now; future extension could introduce a small
           hierarchy if we grow per-piece behavior.
           """
           
           from enum import Enum
           
           class Color(Enum):
               WHITE = 'w'
               BLACK = 'b'
           
               def __str__(self):
                   return self.value
           
           class Piece:
               VALID_KINDS = {'K','Q','R','B','N','P'}
           
               def __init__(self, color: Color, kind: str):
                   if not isinstance(color, Color):
                       raise TypeError("color must be an instance of Color Enum")
                   k = (kind or '').upper()
                   if k not in self.VALID_KINDS:
                       raise ValueError(f"Invalid piece kind: {kind}. Must be one of {sorted(self.VALID_KINDS)}")
                   self.color = color
                   self.kind = k
           
               def __repr__(self):
                   return f"{self.color.value}{self.kind}"
           
               def __str__(self):
                   return self.__repr__()
           
               def __eq__(self, other):
                   if not isinstance(other, Piece):
                       return False
                   return self.color == other.color and self.kind == other.kind
           
               def __hash__(self):
                   return hash((self.color, self.kind))
           
               @property
               def is_white(self) -> bool:
                   return self.color == Color.WHITE
           
               @property
               def is_black(self) -> bool:
                   return self.color == Color.BLACK
           
           __all__ = ['Color', 'Piece']

           """Move generation for Chess MVP (pseudo-legal, no legality checks yet).
           
           This generator traverses the current board and returns a list of Move objects
           representing possible moves for the given color, based on standard piece movement
           patterns and basic blocking rules. It does not yet enforce checks, castling,
           en passant, or promotions beyond a default promotion to 'Q' when applicable.
           """
           
           from typing import List
           
           from .board import Board
           from .move import Move
           from .pieces import Color, Piece
           
           
           class MoveGenerator:
               def __init__(self, board: Board, color: Color):
                   self.board = board
                   self.color = color
           
               def generate_moves(self) -> List[Move]:
                   moves: List[Move] = []
                   for r in range(8):
                       for c in range(8):
                           piece = self.board.get_piece(r, c)
                           if piece is None or piece.color != self.color:
                               continue
                           moves.extend(self._generate_piece_moves(r, c, piece))
                   return moves
           
               def _generate_piece_moves(self, r: int, c: int, piece: Piece) -> List[Move]:
                   kind = piece.kind
                   if kind == 'P':
                       return self._pawn_moves(r, c, piece)
                   if kind == 'N':
                       return self._knight_moves(r, c, piece)
                   if kind == 'B':
                       return self._bishop_moves(r, c, piece)
                   if kind == 'R':
                       return self._rook_moves(r, c, piece)
                   if kind == 'Q':
                       return self._queen_moves(r, c, piece)
                   if kind == 'K':
                       return self._king_moves(r, c, piece)
                   return []
           
               def _pawn_moves(self, r: int, c: int, piece: Piece) -> List[Move]:
                   moves: List[Move] = []
                   dir = -1 if piece.is_white else 1  # white moves up (decreasing row)
                   start_row = 6 if piece.is_white else 1
                   # Forward one
                   fr = r + dir
                   if self._in_bounds(fr, c) and self.board.get_piece(fr, c) is None:
                       to_row = fr
                       promo = 'Q' if to_row == 0 and piece.is_white else None
                       moves.append(Move(r, c, to_row, c, promotion=promo))
                       # Forward two from start position
                       if r == start_row:
                           ffr = r + 2 * dir
                           if self._in_bounds(ffr, c) and self.board.get_piece(ffr, c) is None:
                               moves.append(Move(r, c, ffr, c))
                   # Captures
                   for dc in (-1, 1):
                       cr, cc = r + dir, c + dc
                       if self._in_bounds(cr, cc):
                           target = self.board.get_piece(cr, cc)
                           if target is not None and target.color != piece.color:
                               promo = 'Q' if cr == 0 and piece.is_white else None
                               moves.append(Move(r, c, cr, cc, promotion=promo))
                   return moves
           
               def _knight_moves(self, r: int, c: int, piece: Piece) -> List[Move]:
                   moves: List[Move] = []
                   deltas = [(-2, -1), (-2, 1), (-1, -2), (-1, 2), (1, -2), (1, 2), (2, -1), (2, 1)]
                   for dr, dc in deltas:
                       nr, nc = r + dr, c + dc
                       if not self._in_bounds(nr, nc):
                           continue
                       target = self.board.get_piece(nr, nc)
                       if target is None or target.color != piece.color:
                           moves.append(Move(r, c, nr, nc))
                   return moves
           
               def _bishop_moves(self, r: int, c: int, piece: Piece) -> List[Move]:
                   moves: List[Move] = []
                   for dr, dc in [(-1, -1), (-1, 1), (1, -1), (1, 1)]:
                       nr, nc = r + dr, c + dc
                       while self._in_bounds(nr, nc):
                           target = self.board.get_piece(nr, nc)
                           if target is None:
                               moves.append(Move(r, c, nr, nc))
                           else:
                               if target.color != piece.color:
                                   moves.append(Move(r, c, nr, nc))
                               break
                           nr += dr
                           nc += dc
                   return moves
           
               def _rook_moves(self, r: int, c: int, piece: Piece) -> List[Move]:
                   moves: List[Move] = []
                   for dr, dc in [(-1, 0), (1, 0), (0, -1), (0, 1)]:
                       nr, nc = r + dr, c + dc
                       while self._in_bounds(nr, nc):
                           target = self.board.get_piece(nr, nc)
                           if target is None:
                               moves.append(Move(r, c, nr, nc))
                           else:
                               if target.color != piece.color:
                                   moves.append(Move(r, c, nr, nc))
                               break
                           nr += dr
                           nc += dc
                   return moves
           
               def _queen_moves(self, r: int, c: int, piece: Piece) -> List[Move]:
                   return self._rook_moves(r, c, piece) + self._bishop_moves(r, c, piece)
           
               def _king_moves(self, r: int, c: int, piece: Piece) -> List[Move]:
                   moves: List[Move] = []
                   for dr in (-1, 0, 1):
                       for dc in (-1, 0, 1):
                           if dr == 0 and dc == 0:
                               continue
                           nr, nc = r + dr, c + dc
                           if not self._in_bounds(nr, nc):
                               continue
                           target = self.board.get_piece(nr, nc)
                           if target is None or target.color != piece.color:
                               moves.append(Move(r, c, nr, nc))
                   return moves
           
               @staticmethod
               def _in_bounds(r: int, c: int) -> bool:
                   return 0 <= r < 8 and 0 <= c < 8

                   return []
           
               def _pawn_moves(self, r: int, c: int, piece: Piece) -> List[Move]:
                   moves: List[Move] = []
                   dir = -1 if piece.is_white else 1  # white moves up (decreasing row)
                   start_row = 6 if piece.is_white else 1
                   # Forward one
                   fr = r + dir
                   if self._in_bounds(fr, c) and self.board.get_piece(fr, c) is None:
                       to_row = fr
                       promo = 'Q' if to_row == 0 and piece.is_white else None
                       moves.append(Move(r, c, to_row, c, promotion=promo))
                       # Forward two from start position
                       if r == start_row:
                           ffr = r + 2 * dir
                           if self._in_bounds(ffr, c) and self.board.get_piece(ffr, c) is None:
                               moves.append(Move(r, c, ffr, c))
                   # Captures
                   for dc in (-1, 1):
                       cr, cc = r + dir, c + dc
                       if self._in_bounds(cr, cc):
                           target = self.board.get_piece(cr, cc)
                           if target is not None and target.color != piece.color:
                               promo = 'Q' if cr == 0 and piece.is_white else None
                               moves.append(Move(r, c, cr, cc, promotion=promo))
                   return moves
           
               def _knight_moves(self, r: int, c: int, piece: Piece) -> List[Move]:
                   moves: List[Move] = []
                   deltas = [(-2, -1), (-2, 1), (-1, -2), (-1, 2), (1, -2), (1, 2), (2, -1), (2, 1)]
                   for dr, dc in deltas:
                       nr, nc = r + dr, c + dc
                       if not self._in_bounds(nr, nc):
                           continue
                       target = self.board.get_piece(nr, nc)
                       if target is None or target.color != piece.color:
                           moves.append(Move(r, c, nr, nc))
                   return moves
           
               def _bishop_moves(self, r: int, c: int, piece: Piece) -> List[Move]:
                   moves: List[Move] = []
                   for dr, dc in [(-1, -1), (-1, 1), (1, -1), (1, 1)]:
                       nr, nc = r + dr, c + dc
                       while self._in_bounds(nr, nc):
                           target = self.board.get_piece(nr, nc)
                           if target is None:
                               moves.append(Move(r, c, nr, nc))
                           else:
                               if target.color != piece.color:
                                   moves.append(Move(r, c, nr, nc))
                               break
                           nr += dr
                           nc += dc
                   return moves
           
               def _rook_moves(self, r: int, c: int, piece: Piece) -> List[Move]:
                   moves: List[Move] = []
                   for dr, dc in [(-1, 0), (1, 0), (0, -1), (0, 1)]:
                       nr, nc = r + dr, c + dc
                       while self._in_bounds(nr, nc):
                           target = self.board.get_piece(nr, nc)
                           if target is None:
                               moves.append(Move(r, c, nr, nc))
                           else:
                               if target.color != piece.color:
                                   moves.append(Move(r, c, nr, nc))
                               break
                           nr += dr
                           nc += dc
                   return moves
           
               def _queen_moves(self, r: int, c: int, piece: Piece) -> List[Move]:
                   return self._rook_moves(r, c, piece) + self._bishop_moves(r, c, piece)
           
               def _king_moves(self, r: int, c: int, piece: Piece) -> List[Move]:
                   moves: List[Move] = []
                   for dr in (-1, 0, 1):
                       for dc in (-1, 0, 1):
                           if dr == 0 and dc == 0:
                               continue
                           nr, nc = r + dr, c + dc
                           if not self._in_bounds(nr, nc):
                               continue
                           target = self.board.get_piece(nr, nc)
                           if target is None or target.color != piece.color:
                               moves.append(Move(r, c, nr, nc))
                   return moves
           
               @staticmethod
               def _in_bounds(r: int, c: int) -> bool:
                   return 0 <= r < 8 and 0 <= c < 8

                       line.append(str(piece) if piece else '.')

               print("  a b c d e f g h")

           """Piece model for chess MVP.
           Single responsibility: represent a chess piece with color and kind.
           This is a simple data container now; future extension could introduce a small
           hierarchy if we grow per-piece behavior.
           """
           
           from enum import Enum
           
           class Color(Enum):
               WHITE = 'w'
               BLACK = 'b'
           
               def __str__(self):
                   return self.value
           
           class Piece:
               VALID_KINDS = {'K','Q','R','B','N','P'}
           
               def __init__(self, color: Color, kind: str):
                   if not isinstance(color, Color):
                       raise TypeError("color must be an instance of Color Enum")
                   k = (kind or '').upper()
                   if k not in self.VALID_KINDS:
                       raise ValueError(f"Invalid piece kind: {kind}. Must be one of {sorted(self.VALID_KINDS)}")
                   self.color = color
                   self.kind = k
           
               def __repr__(self):
                   return f"{self.color.value}{self.kind}"
           
               def __str__(self):
                   # Single-character representation: White pieces uppercase, Black pieces lowercase
                   if self.color == Color.WHITE:
                       return self.kind.upper()
                   else:
                       return self.kind.lower()
           
               def __eq__(self, other):
                   if not isinstance(other, Piece):
                       return False
                   return self.color == other.color and self.kind == other.kind
           
               def __hash__(self):
                   return hash((self.color, self.kind))
           
               @property
               def is_white(self) -> bool:
                   return self.color == Color.WHITE
           
               @property
               def is_black(self) -> bool:
                   return self.color == Color.BLACK
           
           __all__ = ['Color', 'Piece']

           """Move legality checks for Chess MVP.
           
           This module enforces real chess legality for a move, including not leaving your
           own king in check. It relies on the existing MoveGenerator for pseudo-legal move
           generation and uses board cloning to simulate moves.
           """
           
           from typing import Optional
           
           from .board import Board
           from .move import Move
           from .pieces import Color
           from .move_generator import MoveGenerator
           
           
           def _find_king(board: Board, color: Color) -> Optional[tuple]:
               for r in range(8):
                   for c in range(8):
                       piece = board.get_piece(r, c)
                       if piece is not None and piece.color == color and piece.kind == 'K':
                           return (r, c)
               return None
           
           
           def is_in_check(board: Board, color: Color) -> bool:
               king_pos = _find_king(board, color)
               if king_pos is None:
                   return False
               kr, kc = king_pos
               attacker = Color.BLACK if color == Color.WHITE else Color.WHITE
               gen = MoveGenerator(board, attacker)
               for m in gen.generate_moves():
                   if m.to_row == kr and m.to_col == kc:
                       return True
               return False
           
           
           def is_move_legal(board: Board, move: Move, color: Color) -> bool:
               piece = board.get_piece(move.from_row, move.from_col)
               if piece is None or piece.color != color:
                   return False
               dest = board.get_piece(move.to_row, move.to_col)
               if dest is not None and dest.color == color:
                   return False
               # Simulate the move on a cloned board
               clone = board.clone()
               moving_piece = clone.get_piece(move.from_row, move.from_col)
               if moving_piece is None:
                   return False
               if move.promotion:
                   moving_piece.kind = move.promotion.upper()
               clone.set_piece(move.to_row, move.to_col, moving_piece)
               clone.set_piece(move.from_row, move.from_col, None)
               return not is_in_check(clone, color)
           
           __all__ = ['is_in_check', 'is_move_legal']

           """Console-based playable chess MVP (two players, no AI)."""
           
           from typing import Optional
           
           from .board import Board
           from .move import Move
           from .move_generator import MoveGenerator
           from .pieces import Color
           from .ui_console import render_board, parse_move_input
           from .move_validator import is_move_legal
           
           
           def _parse_and_normalize_input(user_input: str) -> str:
               s = user_input.strip().replace(' ', '')
               if s.lower() in {'quit', 'exit'}:
                   return s.lower()
               return s
           
           
           def apply_move(board: Board, move: Move, color: Color) -> bool:
               # Basic apply: move piece from source to destination, respecting color
               piece = board.get_piece(move.from_row, move.from_col)
               if piece is None:
                   return False
               if piece.color != color:
                   return False
               # Destination holding a friendly piece is invalid
               dest = board.get_piece(move.to_row, move.to_col)
               if dest is not None and dest.color == piece.color:
                   return False
               # Promotion handling: update kind if provided
               if move.promotion:
                   piece.kind = move.promotion.upper()
               # Perform the move
               board.set_piece(move.to_row, move.to_col, piece)
               board.set_piece(move.from_row, move.from_col, None)
               return True
           
           
           def run_console_game():
               board = Board()
               board.setup_starting_position()
               current_color = Color.WHITE
           
               print("Welcome to Chess MVP (Console). Two players, no AI. Type moves like e2e4 and press Enter.")
               print("Enter 'quit' to exit.")
           
               while True:
                   render_board(board)
                   print(f"Turn: {current_color.name}")
                   user_input = input("Your move (e.g., e2e4): ")

           """Move legality checks for Chess MVP.
           
           This module enforces real chess legality for a move, including not leaving your
           own king in check. It relies on the existing MoveGenerator for pseudo-legal move
           generation and uses board cloning to simulate moves.
           """
           
           from typing import Optional
           
           from .board import Board
           from .move import Move
           from .pieces import Color
           from .move_generator import MoveGenerator
           
           
           def _find_king(board: Board, color: Color) -> Optional[tuple]:
               for r in range(8):
                   for c in range(8):
                       piece = board.get_piece(r, c)
                       if piece is not None and piece.color == color and piece.kind == 'K':
                           return (r, c)
               return None
           
           
           def is_in_check(board: Board, color: Color) -> bool:
               king_pos = _find_king(board, color)
               if king_pos is None:
                   return False
               kr, kc = king_pos
               attacker = Color.BLACK if color == Color.WHITE else Color.WHITE
               gen = MoveGenerator(board, attacker)
               for m in gen.generate_moves():
                   if m.to_row == kr and m.to_col == kc:
                       return True
               return False
           
           
           def is_move_legal(board: Board, move: Move, color: Color) -> bool:
               piece = board.get_piece(move.from_row, move.from_col)
               if piece is None or piece.color != color:
                   return False
               dest = board.get_piece(move.to_row, move.to_col)
               if dest is not None and dest.color == color:
                   return False
               # Simulate the move on a cloned board
               clone = board.clone()
               moving_piece = clone.get_piece(move.from_row, move.from_col)
               if moving_piece is None:
                   return False
               if move.promotion:
                   moving_piece.kind = move.promotion.upper()
               clone.set_piece(move.to_row, move.to_col, moving_piece)
               clone.set_piece(move.from_row, move.from_col, None)
               return not is_in_check(clone, color)
           
           
           def has_legal_moves(board: Board, color: Color) -> bool:
               gen = MoveGenerator(board, color)
               for m in gen.generate_moves():
                   if is_move_legal(board, m, color):
                       return True
               return False
           
           
           def get_game_status(board: Board, color: Color) -> str:
               # Returns 'ongoing', 'checkmate', or 'stalemate'
               in_check = is_in_check(board, color)
               if has_legal_moves(board, color):
                   return 'ongoing'
               return 'checkmate' if in_check else 'stalemate'
           
           __all__ = ['is_in_check', 'is_move_legal', 'has_legal_moves', 'get_game_status']

           """Move generation for Chess MVP (pseudo-legal, no legality checks yet).
           
           This generator traverses the current board and returns a list of Move objects
           representing possible moves for the given color, based on standard piece movement
           patterns and basic blocking rules. It does not yet enforce checks, castling,
           en passant, or promotions beyond a default promotion to 'Q' when applicable.
           """
           
           from typing import List
           
           from .board import Board
           from .move import Move
           from .pieces import Color, Piece
           
           
           class MoveGenerator:
               def __init__(self, board: Board, color: Color):
                   self.board = board
                   self.color = color
           
               def generate_moves(self) -> List[Move]:
                   moves: List[Move] = []
                   for r in range(8):
                       for c in range(8):
                           piece = self.board.get_piece(r, c)
                           if piece is None or piece.color != self.color:
                               continue
                           moves.extend(self._generate_piece_moves(r, c, piece))
                   return moves
           
               def _generate_piece_moves(self, r: int, c: int, piece: Piece) -> List[Move]:
                   kind = piece.kind
                   if kind == 'P':
                       return self._pawn_moves(r, c, piece)
                   if kind == 'N':
                       return self._knight_moves(r, c, piece)
                   if kind == 'B':
                       return self._bishop_moves(r, c, piece)
                   if kind == 'R':
                       return self._rook_moves(r, c, piece)
                   if kind == 'Q':
                       return self._queen_moves(r, c, piece)
                   if kind == 'K':
                       return self._king_moves(r, c, piece)
                   return []
           
               def _pawn_moves(self, r: int, c: int, piece: Piece) -> List[Move]:
                   moves: List[Move] = []
                   dir = -1 if piece.is_white else 1  # white moves up (decreasing row)
                   start_row = 6 if piece.is_white else 1
                   # Forward one
                   fr = r + dir
                   if self._in_bounds(fr, c) and self.board.get_piece(fr, c) is None:
                       to_row = fr
                       promo = None
                       if (to_row == 0 and piece.is_white) or (to_row == 7 and not piece.is_white):
                           promo = 'Q'
                       moves.append(Move(r, c, to_row, c, promotion=promo))
                       # Forward two from start position
                       if r == start_row:
                           ffr = r + 2 * dir
                           if self._in_bounds(ffr, c) and self.board.get_piece(ffr, c) is None:
                               moves.append(Move(r, c, ffr, c))
                   # Captures
                   for dc in (-1, 1):
                       cr, cc = r + dir, c + dc
                       if self._in_bounds(cr, cc):
                           target = self.board.get_piece(cr, cc)
                           if target is not None and target.color != piece.color:
                               promo = None
                               if (cr == 0 and piece.is_white) or (cr == 7 and not piece.is_white):
                                   promo = 'Q'
                               moves.append(Move(r, c, cr, cc, promotion=promo))
                   return moves
           
               def _knight_moves(self, r: int, c: int, piece: Piece) -> List[Move]:
                   moves: List[Move] = []
                   deltas = [(-2, -1), (-2, 1), (-1, -2), (-1, 2), (1, -2), (1, 2), (2, -1), (2, 1)]
                   for dr, dc in deltas:
                       nr, nc = r + dr, c + dc
                       if not self._in_bounds(nr, nc):
                           continue
                       target = self.board.get_piece(nr, nc)
                       if target is None or target.color != piece.color:
                           moves.append(Move(r, c, nr, nc))
                   return moves
           
               def _bishop_moves(self, r: int, c: int, piece: Piece) -> List[Move]:
                   moves: List[Move] = []
                   for dr, dc in [(-1, -1), (-1, 1), (1, -1), (1, 1)]:
                       nr, nc = r + dr, c + dc
                       while self._in_bounds(nr, nc):
                           target = self.board.get_piece(nr, nc)
                           if target is None:
                               moves.append(Move(r, c, nr, nc))
                           else:
                               if target.color != piece.color:
                                   moves.append(Move(r, c, nr, nc))
                               break
                           nr += dr
                           nc += dc
                   return moves
           
               def _rook_moves(self, r: int, c: int, piece: Piece) -> List[Move]:
                   moves: List[Move] = []
                   for dr, dc in [(-1, 0), (1, 0), (0, -1), (0, 1)]:
                       nr, nc = r + dr, c + dc
                       while self._in_bounds(nr, nc):
                           target = self.board.get_piece(nr, nc)
                           if target is None:
                               moves.append(Move(r, c, nr, nc))
                           else:
                               if target.color != piece.color:
                                   moves.append(Move(r, c, nr, nc))
                               break
                           nr += dr
                           nc += dc
                   return moves
           
               def _queen_moves(self, r: int, c: int, piece: Piece) -> List[Move]:
                   return self._rook_moves(r, c, piece) + self._bishop_moves(r, c, piece)
           
               def _king_moves(self, r: int, c: int, piece: Piece) -> List[Move]:
                   moves: List[Move] = []
                   for dr in (-1, 0, 1):
                       for dc in (-1, 0, 1):
                           if dr == 0 and dc == 0:
                               continue
                           nr, nc = r + dr, c + dc
                           if not self._in_bounds(nr, nc):
                               continue
                           target = self.board.get_piece(nr, nc)
                           if target is None or target.color != piece.color:
                               moves.append(Move(r, c, nr, nc))
                   return moves
           
               @staticmethod
               def _in_bounds(r: int, c: int) -> bool:
                   return 0 <= r < 8 and 0 <= c < 8
