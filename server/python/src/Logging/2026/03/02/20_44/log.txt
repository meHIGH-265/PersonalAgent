           ```python
           import math
           import tkinter as tk
           
           
           def circle_grid_cells(cx: float, cy: float, r: float):
               """
               Return a list of (i, j) integer grid coordinates such that the unit square
               [i, i+1] x [j, j+1] intersects the circle centered at (cx, cy) with radius r.
               A square is included if the circle intersects it (including boundary).
               The coordinates refer to the bottom-left corner of each unit square.
               """
               if r < 0:
                   return []
           
               r2 = r * r
               # Range to check: squares that could intersect the circle
               i_min = int(math.floor(cx - r))
               i_max = int(math.floor(cx + r))
               j_min = int(math.floor cy - r)
               j_max = int(math.floor cy + r)
           
               cells = []
               for i in range(i_min, i_max + 1):
                   for j in range(j_min, j_max + 1):
                       # Closest point on the square to the circle center
                       closest_x = min(max(cx, i), i + 1)
                       closest_y = min(max(cy, j), j + 1)
                       dx = cx - closest_x
                       dy = cy - closest_y
                       if dx * dx + dy * dy <= r2 + 1e-12:
                           cells.append((i, j))
               return cells
           
           
           class CircleGridGUI:
               def __init__(self, master):
                   self.master = master
                   self.master.title("Circle Grid Approximation (Squares)")
           
                   # Visualization parameters
                   self.scale = 25  # pixels per world unit
                   self.display_size = 31  # number of grid cells to display horizontally (odd for centering)
                   self.half = self.display_size // 2  # used for centering around the center
           
                   # State
                   self.cx = 0.0
                   self.cy = 0.0
                   self.r = 5.0
           
                   self.dragging = False
                   self.last_x = 0
                   self.last_y = 0
           
                   # Layout frames
                   self.canvas = tk.Canvas(self.master, width=self.display_size * self.scale,
                                           height=self.display_size * self.scale, bg="white")
                   self.canvas.grid(row=0, column=0, rowspan=20, padx=10, pady=10)
           
                   ctrl_frame = tk.Frame(self.master)
                   ctrl_frame.grid(row=0, column=1, sticky="n", padx=10, pady=10)
           
                   # Center X
                   tk.Label(ctrl_frame, text="Center X:").grid(row=0, column=0, sticky="e")
                   self.cx_var = tk.StringVar(value=str(self.cx))
                   self.cx_entry = tk.Entry(ctrl_frame, textvariable=self.cx_var, width=12)
                   self.cx_entry.grid(row=0, column=1, pady=2)
                   self.cx_entry.bind("<Return>", lambda e: self.update_from_entries())
           
                   # Center Y
                   tk.Label(ctrl_frame, text="Center Y:").grid(row=1, column=0, sticky="e")
                   self.cy_var = tk.StringVar(value=str(self.cy))
                   self.cy_entry = tk.Entry(ctrl_frame, textvariable=self.cy_var, width=12)
                   self.cy_entry.grid(row=1, column=1, pady=2)
                   self.cy_entry.bind("<Return>", lambda e: self.update_from_entries())
           
                   # Radius
                   tk.Label(ctrl_frame, text="Radius:").grid(row=2, column=0, sticky="e")
                   self.r_var = tk.StringVar(value=str(self.r))
                   self.r_entry = tk.Entry(ctrl_frame, textvariable=self.r_var, width=12)
                   self.r_entry.grid(row=2, column=1, pady=2)
                   self.r_entry.bind("<Return>", lambda e: self.update_from_entries())
           
                   # Radius slider
                   tk.Label(ctrl_frame, text="Radius (slider):").grid(row=3, column=0, columnspan=2, sticky="w", pady=(8, 0))
                   self.radius_slider = tk.Scale(ctrl_frame, from_=0.0, to=40.0, resolution=0.1,
                                                 orient=tk.HORIZONTAL, length=210, command=self.on_radius_slider)
                   self.radius_slider.set(self.r)
                   self.radius_slider.grid(row=4, column=0, columnspan=2, pady=2)
           
                   # Update button (optional)
                   self.update_btn = tk.Button(ctrl_frame, text="Apply", command=self.update_from_entries)
                   self.update_btn.grid(row=5, column=0, columnspan=2, pady=8)
           
                   # Bind canvas interactions
                   self.canvas.bind("<ButtonPress-1>", self.on_mouse_down)
                   self.canvas.bind("<B1-Motion>", self.on_mouse_move)
                   self.canvas.bind("<ButtonRelease-1>", self.on_mouse_up)
           
                   # Initial draw
                   self.redraw()
           
               def get_view_bounds(self):
                   # Determine a window around the current center to render
                   # We center roughly on floor(cx), floor(cy)
                   view_min_x = int(math.floor(self.cx)) - self.half
                   view_min_y = int(math.floor(self.cy)) - self.half
                   return view_min_x, view_min_y
           
               def world_to_canvas(self, wx, wy):
                   view_min_x, view_min_y = self.get_view_bounds()
                   x = (wx - view_min_x) * self.scale
                   y = (wy - view_min_y) * self.scale
                   return x, y
           
               def canvas_to_world(self, x, y):
                   view_min_x, view_min_y = self.get_view_bounds()
                   wx = view_min_x + x / self.scale
                   wy = view_min_y + y / self.scale
                   return wx, wy
           
               def on_radius_slider(self, val):
                   self.r = float(val)
                   self.r_var.set(str(self.r))
                   self.update_entries_from_state()
                   self.redraw()
           
               def update_from_entries(self):
                   try:
                       new_cx = float(self.cx_var.get())
                       new_cy = float(self.cy_var.get())
                       new_r = float(self.r_var.get())
                   except ValueError:
                       return  # ignore invalid input
                   self.cx = new_cx
                   self.cy = new_cy
                   self.r = new_r
                   # Keep slider in sync
                   self.radius_slider.set(self.r)
                   self.update_entries_from_state()
                   self.redraw()
           
               def update_entries_from_state(self):
                   self.cx_var.set(f"{self.cx:.6g}")
                   self.cy_var.set(f"{self.cy:.6g}")
                   self.r_var.set(f"{self.r:.6g}")
           
               def on_mouse_down(self, event):
                   # Determine if user clicked near the center; if so, start dragging
                   view_min_x, view_min_y = self.get_view_bounds()
                   center_x, center_y = self.world_to_canvas(self.cx, self.cy)
                   dist_sq = (event.x - center_x) ** 2 + (event.y - center_y) ** 2
                   if dist_sq <= (10) ** 2:  # 10 px tolerance
                       self.dragging = True
                       self.last_x = event.x
                       self.last_y = event.y
           
               def on_mouse_move(self, event):
                   if self.dragging:
                       dx = event.x - self.last_x
                       dy = event.y - self.last_y
                       self.cx += dx / self.scale
                       self.cy += dy / self.scale
                       self.last_x = event.x
                       self.last_y = event.y
                       self.update_entries_from_state()
                       self.redraw()
           
               def on_mouse_up(self, event):
                   self.dragging = False
           
               def redraw(self):
                   self.canvas.delete("all")
           
                   view_min_x, view_min_y = self.get_view_bounds()
                   W = self.display_size
                   H = self.display_size
                   # Compute which squares intersect the circle for quick fill
                   circle_cells = set(circle_grid_cells(self.cx, self.cy, self.r))
           
                   # Draw grid and cells
                   for i in range(view_min_x, view_min_x + W):
                       for j in range(view_min_y, view_min_y + H):
                           left = (i - view_min_x) * self.scale
                           top = (j - view_min_y) * self.scale
                           right = left + self.scale
                           bottom = top + self.scale
           
                           if (i, j) in circle_cells:
                               fill = "#87CEFA"  # light blue for inside-circle squares
                           else:
                               fill = "white"
           
                           self.canvas.create_rectangle(left, top, right, bottom, fill=fill, outline="#cccccc")
           
                   # Draw circle outline for comparison
                   cx_pix, cy_pix = self.world_to_canvas(self.cx, self.cy)
                   r_pix = self.r * self.scale
                   self.canvas.create_oval(cx_pix - r_pix, cy_pix - r_pix,
                                           cx_pix + r_pix, cy_pix + r_pix,
                                           outline="#0000ff", width=2)
           
                   # Draw center point
                   self.canvas.create_oval(cx_pix - 4, cy_pix - 4, cx_pix + 4, cy_pix + 4,
                                           fill="#ff0000", outline="")
           
               def update(self):
                   # Convenience method if you want to trigger a redraw from outside
                   self.update_entries_from_state()
                   self.redraw()
           
           
           if __name__ == "__main__":
               root = tk.Tk()
               app = CircleGridGUI(root)
               root.mainloop()
           ```
