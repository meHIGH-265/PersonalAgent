           """Simple AI logic for the MVP."""
           
           from __future__ import annotations
           
           import random
           from typing import Optional, Dict
           
           from .config import MAP_TERRITORY_COUNT
           from .models import Territory, GameState
           
           
           def decide_ai_action(game_state: GameState, ai_player_id: int) -> Optional[dict]:
               """Decide a single action for an AI player.
           
               Strategy (simple and deterministic for MVP):
                 - If an owned territory has an adjacent enemy territory and has more units, attack it.
                 - Otherwise, recruit in a territory that has the fewest units.
               Returns an action dict or None if no action is chosen.
               """
               territories = game_state.territories
               ai_player = game_state.players.get(ai_player_id)
               if ai_player is None:
                   return None
           
               # Try to find an attack opportunity
               for tid in list(ai_player.territories):
                   t = territories.get(tid)
                   if t is None:
                       continue
                   for nb in t.adjacent_ids:
                       nb_t = territories.get(nb)
                       if nb_t and nb_t.owner is not None and nb_t.owner != ai_player_id:
                           # If we have more units than the defender, attack
                           if t.units > nb_t.units:
                               return {"player_id": ai_player_id, "action": "ATTACK", "src": tid, "dst": nb}
           
               # Otherwise recruit in a territory with the fewest units
               if ai_player.territories:
                   best_territory = min(ai_player.territories, key=lambda x: territories[x].units)
                   return {"player_id": ai_player_id, "action": "RECRUIT", "territory": best_territory}
           
               return None
           
           
           __all__ = ["decide_ai_action"]

           """Core game engine for the MVP (tick-based with cooldowns)."""
           
           from __future__ import annotations
           
           import math
           from typing import Dict, List, Optional
           
           from .config import MAP_TERRITORY_COUNT
           from .models import Territory, GameState, Player, ActionState
           from .ai import decide_ai_action
           from .config import PRODUCTION_PER_TICK
           
           
           class GameEngine:
               def __init__(self, territories: Dict[int, Territory], players: Dict[int, Player]):
                   self.territories = territories
                   self.players = players
                   self.state = GameState(territories=territories, players=players)
                   self.state.time = 0.0
                   self._init_action_states()
           
               def _init_action_states(self) -> None:
                   self.state.action_states = {pid: [] for pid in self.players.keys()}
           
               def is_game_over(self) -> bool:
                   # Win condition: a single player owns all territories
                   owners = {t.owner for t in self.territories.values()}
                   owners.discard(None)
                   return len(owners) <= 1 and len(owners) == 1 or any(t.owner is None for t in self.territories.values()) is False
           
               def get_winner(self) -> Optional[int]:
                   owners = {t.owner for t in self.territories.values() if t.owner is not None}
                   if len(owners) == 1:
                       return next(iter(owners))
                   return None
           
               def tick(self, delta: float) -> None:
                   self.state.time += delta
                   # Production: each owned territory increases units by production
                   for t in self.territories.values():
                       t.units += t.production
                       if t.units < 0:
                           t.units = 0
           
                   # Decrease cooldowns for all players
                   for pid, actions in self.state.action_states.items():
                       for a in actions:
                           a.cooldown_remaining = max(0.0, a.cooldown_remaining - delta)
           
                   # AI actions
                   for pid, player in self.players.items():
                       if player.is_ai:
                           action = decide_ai_action(self.state, pid)
                           if action:
                               self._apply_action(action)
           
                   # Check win condition after AI actions
                   winner = self.get_winner()
                   if winner is not None:
                       self.state.winner = winner
           
               def _ensure_action_state(self, pid: int, action_type: str, cooldown: float) -> None:
                   lst = self.state.action_states.setdefault(pid, [])
                   # If there is an existing action of this type with cooldown > 0, ignore; otherwise add
                   for a in lst:
                       if a.action_type == action_type:
                           a.cooldown_remaining = max(a.cooldown_remaining, cooldown)
                           return
                   lst.append(ActionState(action_type=action_type, cooldown_remaining=cooldown))
           
               def can_perform(self, pid: int, action_type: str) -> bool:
                   actions = self.state.action_states.get(pid, [])
                   for a in actions:
                       if a.action_type == action_type:
                           return a.cooldown_remaining <= 0.0
                   return True
           
               def _apply_action(self, action: dict) -> None:
                   # action dict: {"player_id": int, "action": str, ...}
                   player_id = action["player_id"]
                   act = action["action"]
                   if act == "ATTACK":
                       self._attack(action["src"], action["dst"], player_id)
                       return
                   if act == "MOVE":
                       self._move(action["src"], action["dst"], player_id)
                       return
                   if act == "RECRUIT":
                       self._recruit(action["territory"], player_id)
                       return
           
               def _move(self, src: int, dst: int, pid: int) -> None:
                   s = self.territories.get(src)
                   d = self.territories.get(dst)
                   if not s or not d:
                       return
                   if s.owner != pid:
                       return
                   if dst not in s.adjacent_ids:
                       return
                   if s.units <= 0:
                       return
                   if d.owner != pid:
                       return
                   # Move one unit from src to dst if possible
                   s.units -= 1
                   d.units += 1
           
               def _attack(self, src: int, dst: int, pid: int) -> None:
                   s = self.territories.get(src)
                   d = self.territories.get(dst)
                   if not s or not d:
                       return
                   if s.owner != pid:
                       return
                   if dst not in s.adjacent_ids:
                       return
                   if s.units <= 0:
                       return
                   # Simple combat: attacker wins if has more units than defender
                   attacker = s.units
                   defender = d.units
                   # Always use at least 1 attack unit
                   s.units = max(0, s.units - 1)
                   if attacker > defender:
                       # Conquer territory
                       d.owner = pid
                       d.units = max(1, attacker - defender)
                   else:
                       # Defender wins, lose some units
                       d.units = max(0, defender - (attacker // 2))
           
               def _recruit(self, territory_id: int, pid: int) -> None:
                   t = self.territories.get(territory_id)
                   if not t:
                       return
                   if t.owner != pid:
                       return
                   t.units += 1
           
           
           __all__ = ["GameEngine"]

           """Entrypoint for the MVP CLI game."""
           
           from __future__ import annotations
           
           import threading
           import queue
           import time
           
           from .config import MAP_TERRITORY_COUNT, MAP_TERRITORY_NAMES, TICK_SECONDS
           from .map import build_initial_map
           from .models import Territory, GameState, Player

           def main() -> None:
               # Build initial map
               territories = build_initial_map()
               # Create players
               human = Player(id=1, name="You", color="blue", is_ai=False)
               ai = Player(id=2, name="AI", color="red", is_ai=True)
               players = {human.id: human, ai.id: ai}
           
               # Assign initial territories: split ring evenly
               terr_ids = list(territories.keys())
               half = len(terr_ids) // 2
               for tid in terr_ids[:half]:
                   territories[tid].owner = human.id
                   human.territories.add(tid)
               for tid in terr_ids[half:]:
                   territories[tid].owner = ai.id
                   ai.territories.add(tid)
           
               # Create game engine
               engine = GameEngine(territories, players)
           
               # Command queue for human input
               cmd_queue: queue.Queue[str] = queue.Queue()
           
               def input_thread() -> None:
                   while True:
                       try:
                           line = input("Enter command (MOVE src dst | ATTACK src dst | RECRUIT territory): ")
                           if line:
                               cmd_queue.put(line.strip())
                       except EOFError:
                           break
           
               t = threading.Thread(target=input_thread, daemon=True)
               t.start()
           
               # Main loop
               try:
                   while True:
                       # Tick the engine
                       engine.tick(TICK_SECONDS)
                       # Render state
                       render_state(engine.state)
                       render_players(engine.players)
                       # Process human input if any
                       try:
                           line = cmd_queue.get_nowait()
                           parts = line.upper().split()
                           if not parts:
                               continue
                           cmd = parts[0]

                   owners = {t.owner for t in self.territories.values()}

                           t.units = 0

                           action = decide_ai_action(self.state, pid)
                           if action:
                               self._apply_action(action)

                   if winner is not None:

                       if t.units < 0:
                           t.units = 0
           
                   # Decrease cooldowns for all players
                   for pid, actions in self.state.action_states.items():
                       for a in actions:
                           a.cooldown_remaining = max(0.0, a.cooldown_remaining - delta)
           
                   # AI actions
                   for pid, player in self.players.items():
                       if player.is_ai:
                           action = decide_ai_action(self.state, pid)
                           if action:
                               self._apply_action(action)
           
                   # Check win condition after AI actions
                   winner = self.get_winner()
                   if winner is not None:
                       self.state.winner = winner
           
               def _ensure_action_state(self, pid: int, action_type: str, cooldown: float) -> None:
                   lst = self.state.action_states.setdefault(pid, [])
                   # If there is an existing action of this type with cooldown > 0, ignore; otherwise add
                   for a in lst:
                       if a.action_type == action_type:
                           a.cooldown_remaining = max(a.cooldown_remaining, cooldown)
                           return
                   lst.append(ActionState(action_type=action_type,
                              cooldown_remaining=cooldown))
           
               def can_perform(self, pid: int, action_type: str) -> bool:
                   actions = self.state.action_states.get(pid, [])
                   for a in actions:
                       if a.action_type == action_type:
                           return a.cooldown_remaining <= 0.0
                   return True
           
               def _apply_action(self, action: dict) -> None:
                   # action dict: {"player_id": int, "action": str, ...}
                   player_id = action["player_id"]
                   act = action["action"]
                   if act == "ATTACK":
                       if not self.can_perform(player_id, "ATTACK"):
                           return
                       self._attack(action["src"], action["dst"], player_id)
                       self._start_cooldown(player_id, "ATTACK")
                       return
                   if act == "MOVE":
                       if not self.can_perform(player_id, "MOVE"):
                           return
                       self._move(action["src"], action["dst"], player_id)
                       self._start_cooldown(player_id, "MOVE")
                       return
                   if act == "RECRUIT":
                       if not self.can_perform(player_id, "RECRUIT"):
                           return
                       self._recruit(action["territory"], player_id)
                       self._start_cooldown(player_id, "RECRUIT")
                       return
           
               def _start_cooldown(self, pid: int, action_type: str) -> None:
                   cooldown = default_cooldown(ActionType[action_type]) if action_type in ActionType.__members__ else 2.0
                   self._ensure_action_state(pid, action_type, cooldown)
           
               def perform_action(self, action: dict) -> bool:
                   """Public wrapper to perform an action via GUI or external callers.
                   Returns True if the action was accepted and applied, False otherwise.
                   """
                   pid = action.get("player_id")
                   act = action.get("action")
                   if pid is None or act is None:
                       return False
                   if not self.can_perform(pid, act):
                       return False
                   self._apply_action(action)
                   return True
           
               def _move(self, src: int, dst: int, pid: int) -> None:
                   s = self.territories.get(src)
                   d = self.territories.get(dst)
                   if not s or not d:
                       return
                   if s.owner != pid:
                       return
                   if dst not in s.adjacent_ids:
                       return
                   if s.units <= 0:
                       return
                   if d.owner != pid:
                       return
                   # Move one unit from src to dst if possible
                   s.units -= 1
                   d.units += 1
           
               def _attack(self, src: int, dst: int, pid: int) -> None:
                   s = self.territories.get(src)
                   d = self.territories.get(dst)
                   if not s or not d:
                       return
                   if s.owner != pid:
                       return
                   if dst not in s.adjacent_ids:
                       return
                   if s.units <= 0:
                       return
                   # Simple combat: attacker wins if has more units than defender
                   attacker = s.units
                   defender = d.units
                   # Always use at least 1 attack unit
                   s.units = max(0, s.units - 1)
                   if attacker > defender:
                       # Conquer territory
                       d.owner = pid
                       d.units = max(1, attacker - defender)
                   else:
                       # Defender wins, lose some units
                       d.units = max(0, defender - (attacker // 2))
           
               def _recruit(self, territory_id: int, pid: int) -> None:
                   t = self.territories.get(territory_id)
                   if not t:
                       return
                   if t.owner != pid:
                       return
                   t.units += 1

           from __future__ import annotations

           # Visual config

               if owner_id is None:

                   pygame.draw.circle(screen, COLOR_BORDER, pos, TERRITORY_RADIUS, 2)
                   # label and units

                   text = font.render(label, True, TEXT_COLOR)
                   text_rect = text.get_rect(center=(pos[0], pos[1] - 20))

                   name_text = font.render(t.name, True, TEXT_COLOR)
                   name_rect = name_text.get_rect(center=(pos[0], pos[1] + 30))
                   screen.blit(name_text, name_rect)
                   if t.owner is None:
                       owner_text = font.render("N", True, (200, 200, 200))
                   else:
                       owner_text = font.render(str(t.owner), True, TEXT_COLOR)
                   owner_rect = owner_text.get_rect(center=(pos[0], pos[1] + 50))
                   screen.blit(owner_text, owner_rect)

                   for idx, tid in enumerate(sorted(territories.keys()), start=0):
                       tpos = _territory_position(idx)
                       dx = posx - tpos[0]
                       dy = posy - tpos[1]
                       if dx * dx + dy * dy <= TERRITORY_RADIUS * TERRITORY_RADIUS:
                           return tid
                   return None
           
               while running:
                   delta = clock.tick(FPS) / 1000.0  # seconds since last frame
                   # Let engine advance in real-time in fixed steps for stability
                   # We accumulate time and tick in fixed increments if needed
                   # For simplicity we tick every frame with a small delta
                   engine.tick(delta)
           
                   # Render
                   screen.fill(COLOR_BG)
                   # Draw territories in a circular layout
                   for idx, tid in enumerate(sorted(territories.keys()), start=0):
                       t = territories[tid]
                       draw_territory(t, idx)
           
                   # Draw HUD
                   hud_text = f"Time {engine.state.time:.2f}s - You: {human.territories} - AI: {ai.territories}"
                   hud_surf = font.render(hud_text, True, TEXT_COLOR)
                   screen.blit(hud_surf, (10, 10))
           
                   # Selected indicator
                   if selected_src is not None:
                       pos = _territory_position([k for k in range(1, MAP_TERRITORY_COUNT + 1)].index(selected_src))
                       pygame.draw.circle(screen, (255, 255, 0), pos, TERRITORY_RADIUS + 6, 2)
           
                   pygame.display.flip()
           
                   # Event handling
                   for event in pygame.event.get():
                       if event.type == pygame.QUIT:
                           running = False
                       elif event.type == pygame.MOUSEBUTTONDOWN and event.button == 1:
                           mx, my = event.pos
                           tid = get_territory_at_pos(mx, my)
                           if tid is None:
                               continue
                           t = territories[tid]
                           # Basic interaction model:
                           # - If human owns the territory, select as source for potential MOVE/ATTACK or RECRUIT
                           if t.owner == human.id:
                               selected_src = tid if selected_src != tid else None
                           else:
                               # If a source is selected, attempt an action from selected_src to tid
                               if selected_src is not None:
                                   # If same owner -> MOVE, if different -> ATTACK
                                   if territories[selected_src].owner == human.id:
                                       if territories[tid].owner == human.id:
                                           engine.perform_action({"player_id": human.id, "action": "MOVE", "src": selected_src, "dst": tid})
                                       else:
                                           engine.perform_action({"player_id": human.id, "action": "ATTACK", "src": selected_src, "dst": tid})
                                   selected_src = None
           
                   # Check for win condition
                   if engine.state.winner is not None:
                       winner_id = engine.state.winner
                       print(f"Winner: Player {winner_id}")
                       time.sleep(2)
                       running = False
           
               pygame.quit()
           
           
           __all__ = ["run_gui"]

                           line = input(

                   while True:
                       # Tick the engine

                       # Render state
                       render_state(engine.state)
                       render_players(engine.players)
                       # Process human input if any
                       try:
                           line = cmd_queue.get_nowait()
                           parts = line.upper().split()
                           if not parts:
                               continue
                           cmd = parts[0]
                           if cmd == "MOVE" and len(parts) == 3:
                               src = int(parts[1])
                               dst = int(parts[2])
                               engine._apply_action(

                               engine._apply_action(
                                   {"player_id": 1, "action": "RECRUIT", "territory": t_id})
                           else:
                               print("Invalid command.")
                       except queue.Empty:
                           pass
           
                       # Check win condition
                       if engine.state.winner is not None:
                           print(f"Player {engine.state.winner} has won the game!")
